cmake_minimum_required(VERSION 3.16)
project(go1_motor_sdk LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
add_library(go1_motor STATIC src/go1_motor.c stm32/go1_transport.c)
target_include_directories(go1_motor PUBLIC include stm32)
if(UNIX)
 target_link_libraries(go1_motor PUBLIC m)
 add_library(go1_serial STATIC src/go1_serial.cpp)
 target_include_directories(go1_serial PUBLIC include)
 target_link_libraries(go1_serial PUBLIC go1_motor)
 add_executable(go1_stop examples/stop.cpp)
 target_link_libraries(go1_stop PRIVATE go1_serial)
endif()
enable_testing()
add_executable(test_protocol tests/test_protocol.c)
target_link_libraries(test_protocol PRIVATE go1_motor)
add_test(NAME protocol COMMAND test_protocol)
